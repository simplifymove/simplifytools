#   Ubuntu/Debian: sudo apt-get install -y imagemagick imagemagick-6-common imagemagick-6.q16 potrace
#   CentOS/RHEL: sudo yum install -y ImageMagick potrace
#   macOS: brew install imagemagick potrace
#   Windows: Download from https://potrace.sourceforge.net/
#
# For video processing & GIF creation:
#   Ubuntu/Debian: sudo apt-get install -y ffmpeg
#   CentOS/RHEL: sudo yum install -y ffmpeg
#
# For OCR support (Tesseract):
#   Ubuntu/Debian: sudo apt-get install -y tesseract-ocr libtesseract-dev
#   CentOS/RHEL: sudo yum install -y tesseract
#
# For additional utilities:
#   Ubuntu/Debian: sudo apt-get install -y ghostscript fonts-dejavu libfreetype6
#   CentOS/RHEL: sudo yum install -y ghostscript dejavu-fonts freetype-devel

#    python3 -m venv .venv
